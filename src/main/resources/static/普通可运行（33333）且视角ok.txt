<div id="cesiumContainer" style="width:100%;height:100vh;"></div>
<div id="infoPanel">
    <div>总车辆数: <span id="totalCount">0</span></div>
    <div>方向1车辆: <span id="direction1Count">0</span></div>
    <div>方向2车辆: <span id="direction2Count">0</span></div>
    <div id="status">状态: 连接中...</div>

    <div class="control-group">
        <label for="vehicleList">选择车辆:</label>
        <select id="vehicleList">
            <option value="">-- 请选择车辆 --</option>
        </select>
        <button id="refreshBtn">刷新列表</button>
    </div>
</div>

<script>
    // 初始化Cesium
    Cesium.Ion.defaultAccessToken = 'eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJqdGkiOiI4Y2FkYmZlNC0zZTFhLTQzZjgtOTAxMC04ODQ1YWE0NTRiZjAiLCJpZCI6MjE1MzczLCJpYXQiOjE3MTU3NjYwMTR9.mpRLVbgOWVmBAtgdpU68N1S6_UnXYWyALKB_iEwrKf8';

    const viewer = new Cesium.Viewer('cesiumContainer', {
        terrainProvider: Cesium.createWorldTerrain(),
        timeline: false,
        animation: false
    });

    // 设置初始视角 - 修改为武汉视角，高度5000米
    viewer.camera.flyTo({
        destination: Cesium.Cartesian3.fromDegrees(114.03636, 30.84627, 5000)
    });

    // 车辆实体存储：车牌号 -> Cesium实体
    const vehicleEntities = new Map();
    let lastUpdateTime = 0;
    let connectionErrorCount = 0;
    let isPaused = false;

    // DOM元素
    const totalDisplay = document.getElementById('totalCount');
    const dir1Display = document.getElementById('direction1Count');
    const dir2Display = document.getElementById('direction2Count');
    const statusDisplay = document.getElementById('status');
    const vehicleList = document.getElementById('vehicleList');
    const refreshBtn = document.getElementById('refreshBtn');

    // 事件监听
    vehicleList.addEventListener('change', focusVehicle);
    refreshBtn.addEventListener('click', refreshVehicleList);

    // 初始化
    updateStatus('正在加载...');
    fetchVehicleData();

    // 启动定时更新
    const updateInterval = setInterval(() => {
        if (!isPaused) fetchVehicleData();
    }, 500);

    // 获取车辆数据
    async function fetchVehicleData() {
        try {
            const [points, numbers] = await Promise.all([
                fetchWithTimeout('http://100.65.38.139:33333/points', 3000),
                fetchWithTimeout('http://100.65.38.139:33333/number', 3000)
            ]);

            if (!points.ok || !numbers.ok) {
                throw new Error(`API响应错误: points ${points.status}, numbers ${numbers.status}`);
            }

            const pointsData = await points.json();
            const numbersData = await numbers.json();

            updatePoints(pointsData);
            updateNumbers(numbersData);
            connectionErrorCount = 0;
            updateStatus(`最后更新: ${new Date().toLocaleTimeString()}`);

        } catch (error) {
            handleFetchError(error);
        }
    }

    // 带超时的fetch
    function fetchWithTimeout(url, timeout) {
        return Promise.race([
            fetch(url),
            new Promise((_, reject) =>
                setTimeout(() => reject(new Error('请求超时')), timeout)
            )
        ]);
    }

    // 更新点数据（增量更新）
    function updatePoints(points) {
        const startTime = Date.now();
        const newVehicleIds = new Set();

        // 更新或新增车辆实体
        points.forEach(point => {
            const entityId = `${point.name}_${point.id}`;
            newVehicleIds.add(entityId);

            if (!vehicleEntities.has(entityId)) {
                // 创建新实体
                const entity = viewer.entities.add(createVehicleEntity(point));
                vehicleEntities.set(entityId, entity);
            } else {
                // 更新现有实体位置
                const entity = vehicleEntities.get(entityId);
                entity.position = Cesium.Cartesian3.fromDegrees(
                    point.longitude, point.latitude
                );

                // 更新标签显示逻辑
                entity.label.show = new Cesium.CallbackProperty(() => {
                    const cameraHeight = viewer.camera.positionCartographic.height;
                    return cameraHeight <= 5000;
                }, false);
            }
        });

        // 移除消失的车辆
        vehicleEntities.forEach((entity, id) => {
            if (!newVehicleIds.has(id)) {
                viewer.entities.remove(entity);
                vehicleEntities.delete(id);
            }
        });

        console.log(`更新${points.length}辆车，耗时${Date.now() - startTime}ms`);
    }

    // 创建车辆实体
    function createVehicleEntity(point) {
        return {
            position: Cesium.Cartesian3.fromDegrees(point.longitude, point.latitude),
            point: {
                pixelSize: 8,
                color: point.direction === 1 ? Cesium.Color.RED : Cesium.Color.BLUE,
                outlineColor: Cesium.Color.WHITE,
                outlineWidth: 2,
                disableDepthTestDistance: Number.POSITIVE_INFINITY
            },
            label: {
                text: `${point.name} | ${point.id}`,
                font: '14px sans-serif',
                pixelOffset: new Cesium.Cartesian2(0, 20),
                show: new Cesium.CallbackProperty(() => {
                    const cameraHeight = viewer.camera.positionCartographic.height;
                    return cameraHeight <= 5000;
                }, false),
                disableDepthTestDistance: Number.POSITIVE_INFINITY
            }
        };
    }

    // 更新数字统计
    function updateNumbers(data) {
        totalDisplay.textContent = data.total;
        dir1Display.textContent = data.direction1;
        dir2Display.textContent = data.direction2;
    }

    // 刷新车辆列表
    async function refreshVehicleList() {
        try {
            const response = await fetch('http://100.65.38.139:33333/points');
            const points = await response.json();

            // 清空现有列表
            vehicleList.innerHTML = '<option value="">-- 请选择车辆 --</option>';

            // 添加新选项
            points.forEach((point, index) => {
                const option = document.createElement('option');
                option.value = point.name;
                option.textContent = `${index + 1}. ${point.name} (${point.direction === 1 ? '方向1' : '方向2'})`;
                vehicleList.appendChild(option);
            });

            updateStatus('车辆列表已刷新');

        } catch (error) {
            updateStatus(`刷新失败: ${error.message}`);
        }
    }

    // 聚焦选定车辆
    function focusVehicle() {
        const selectedPlate = vehicleList.value;
        if (!selectedPlate) return;

        // 在现有车辆中查找
        for (const [id, entity] of vehicleEntities) {
            if (id.startsWith(selectedPlate)) {
                viewer.camera.flyTo({
                    destination: entity.position.getValue(Cesium.JulianDate.now()),
                    offset: new Cesium.HeadingPitchRange(0, Cesium.Math.toRadians(-30), 500),
                    duration: 1
                });
                return;
            }
        }

        updateStatus('车辆当前不可见');
    }

    // 处理错误
    function handleFetchError(error) {
        console.error('数据获取失败:', error);
        connectionErrorCount++;

        if (connectionErrorCount > 3) {
            updateStatus('严重错误: 无法连接服务器');
            isPaused = true;
        } else {
            updateStatus(`网络错误: ${error.message}`);
        }
    }

    // 更新状态显示
    function updateStatus(message) {
        statusDisplay.textContent = `状态: ${message}`;
        statusDisplay.className = message.includes('错误') ? 'error' : '';
    }

    // 清理资源
    window.addEventListener('beforeunload', () => {
        clearInterval(updateInterval);
        viewer.destroy();
    });
</script>
